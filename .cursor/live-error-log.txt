    $mysqli = mysqli_connect($host, $user, $password, $database);

        http_response_code(500);
        exit('Database connection unavailable.');
    }
    mysqli_set_charset($mysqli, 'utf8mb4');
    $GLOBALS['conn'] = $mysqli;

    return $mysqli;
}
